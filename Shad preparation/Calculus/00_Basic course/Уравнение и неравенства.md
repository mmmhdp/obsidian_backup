# <mark class="hltr-coler">Иррациональные уравнения и неравенства</mark>

$$
\sqrt{f(x)} = \sqrt{g(x)} \iff 
\begin{cases}
f(x) = g(x) \\
f(x) \geq 0
\end{cases} \iff
\begin{cases}
f(x) = g(x) \\
g(x) \geq 0
\end{cases}
$$
$$
\sqrt[2k+1]{f(x)} \; \vee \; g(x) \iff f(x)  \; \vee \; (g(x))^{2k+1}, \vee \in \{=,>,<,\leq,\geq,\neq\}
$$
$$
\sqrt[2k]{f(x)} = g(x) \iff 
\begin{cases}
f(x) = (g(x))^{2k}\\
g(x) \geq0
\end{cases}
$$
$$
\sqrt[2k]{f(x)} > g(x) \iff 
\left[
\begin{align}
&\begin{cases}
f(x) \geq 0 \\
g(x) < 0
\end{cases}
\\
&\begin{cases}
f(x) > (g(x))^{2k}\\
g(x) \geq 0
\end{cases}
\end{align}
\right.
$$
$$
\sqrt[2k]{f(x)} \leq g(x) \iff
\begin{cases}
g(x)\geq 0\\
f(x)\geq 0\\
f(x) \leq (g(x))^{2k}
\end{cases}
$$
# <mark class="hltr-coler">Показательные  и логарифмические</mark>
$$
a^{(f(x))} - a^{(g(x))} \vee 0 \iff (a-1)(f(x) -g(x)) \vee 0
$$
$$
\log_{a}f(x) - \log_{a}g(x) \vee 0 \iff (a-1)(f(x) -g(x)) \vee 0
$$
