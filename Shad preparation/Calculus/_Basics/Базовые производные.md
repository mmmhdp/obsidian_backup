# <mark class="hltr-coler">Таблица производных</mark> 
$$
\begin{align}
c'                  & = 0, \; c=const\\
(x^n)'              & = nx^{n-1}\\
(a^x)'              & = a^x \ln a\\
(e^x)'              & = e^x \\
(\log_a{x})'        & = \frac{1}{x\ln a} \\
(\sin x)'           & = \cos x\\
(\cos x)'           & = - \sin x \\
(\text{tg } x)'     & = \frac{1}{\cos^2 x} \\
(\text{ctg } x)'    & = -\frac{1}{\sin^2 x} \\
(\text{arcsin } x)' & = \frac{1}{\sqrt{1-x^2}} \\
(\text{arccos } x)' & = -\frac{1}{\sqrt{1-x^2}}\\
(\text{arctg } x)'  & = \frac{1}{1+x^2}\\
(\text{arcctg } x)' & =  -\frac{1}{1+x^2}\\
(\text{sh } x)'     & = \text{ch } x\\
(\text{ch } x)'     & = \text{sh } x\\
(\text{th } x)'     & = \frac{1}{\text{ch }^2 x} \\
(\text{th } x)'     & = -\frac{1}{\text{sh }^2 x}\\
\end{align}
$$
