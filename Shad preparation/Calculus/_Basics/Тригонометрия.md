# <mark class="hltr-coler">Основное тригонометрическое тождество</mark>
$$
\begin{align}
\sin^2{x} + \cos^2{x} = 1 \\
\text{tg}^2{x} + 1 = \frac{1}{\cos^2{x}}\\
1 + \text{ctg}^2{x} = \frac{1}{\sin^2{x}}
\end{align}
$$
# <mark class="hltr-coler">Формулы сложения</mark>
$$
\begin{align}
&\sin(x+y) = \sin(x)\cos{y} + \cos{x}\sin{y} \\
&\sin(x-y) = \sin(x)\cos{y} - \cos{x}\sin{y} \\
\\
&\cos{(x+y)} = \cos{x}\cos{y} - \sin{x}\sin{y}\\
&\cos{(x+y)} = \cos{x}\cos{y} + \sin{x}\sin{y}\\
\\
&\text{tg }(x+y) = \frac{\text{tg }{x} + \text{tg }{y}}{1-\text{tg }x\text{ tg }y}\\
&\text{tg }(x-y) = \frac{\text{tg }{x} - \text{tg }{y}}{1+\text{tg }x\text{ tg }y}\\
\\
&\text{ctg }(x+y) = \frac{\text{ctg }{x}\text{ ctg }{y}-1}{\text{ctg }{x}+\text{ctg }{y}}\\
&\text{ctg }(x+y) = \frac{\text{ctg }{x}\text{ ctg }{y}-1}{\text{ctg }{x}+\text{ctg }{y}}
\end{align}
$$

# <mark class="hltr-coler">Формулы двойного угла</mark>

$$
\begin{align}
&\sin{2x} = 2\sin{x}\cos{x}\\
\\
&\cos{2x} = \cos^2{x} - \sin^2{x}\\ 
&\cos{2x} = 2\cos^2{x}-1\\
&\cos{2x} = 1- 2\sin^2{x}\\
\\
&\text{tg }{2x} = \frac{2\text{tg }x}{1-\text{tg }^2x}\\
&\text{ctg }{2x} = \frac{\text{ctg }^2x - 1}{2\text{ctg }x}\\
\end{align}
$$
# <mark class="hltr-coler">Формулы тройного угла</mark>
$$
\begin{align}
&\sin{3x} = 3\sin{x} - 4\sin^3{x}\\
&\cos{3x} = 4\cos^3{x}-3\cos{x}\\
&\text{tg }{3x} = \frac{3\text{tg }{x} - \text{tg }^3{x}}{1 - 3\text{tg }^2{x}}
\end{align}
$$
# <mark class="hltr-coler">Формулы понижения степени</mark>
$$
\begin{align}
\cos^2{x} = \frac{1+\cos{2x}}{2} \\
\sin^2{x} = \frac{1-\cos{2x}}{2}
\end{align}
$$
# <mark class="hltr-coler">Формулы тангенса половинного угла</mark>
$$
\begin{align}
\text{tg }^2\frac{x}{2} = \frac{1-\cos{x}}{1+\cos{x}} \\
\text{tg }\frac{x}{2} = \frac{1-\cos{x}}{\sin{x}} \\
\text{tg }\frac{x}{2} = \frac{\sin{x}}{1+\cos{x}}
\end{align}
$$
# <mark class="hltr-coler">Универсальная тригонометрическая подстановка</mark>
$$
\begin{align}
& T = \text{tg }\frac{x}{2}\\
&\sin{x} = \frac{2T}{1+T^2} \\
&\cos{x} = \frac{1-T^2}{1+T^2} \\
&\text{tg }{x} = \frac{2T}{1-T^2}\\
&\text{ctg }{x} = \frac{1-T^2}{2T}
\end{align}
$$
# <mark class="hltr-coler">Суммы и произведения тригонометрических функций</mark>
$$
\begin{align}
\sin{x} + \sin{y} = 2\sin{\left(\frac{x+y}{2}\right)}\cos{\left(\frac{x-y}{2}\right)}\\
\sin{x} - \sin{y} = 2\sin{\left(\frac{x-y}{2}\right)}\cos{\left(\frac{x+y}{2}\right)} \\
\cos{x} + \cos{y} = 2 \cos{\left(\frac{x+y}{2}\right)}\cos{\left(\frac{x-y}{2}\right)}\\
\cos{x} - \cos{y} = 2 \sin{\left(\frac{x+y}{2}\right)}\cos{\left(\frac{y-x}{2}\right)}\\

\text{tg } x + \text{tg } y = \frac{\sin(x + y)}{\cos x \cos y}\\
\text{tg } x - \text{tg } y = \frac{\sin(x - y)}{\cos x \cos y}\\
\text{ctg } x + \text{ctg } y = \frac{\sin(x + y)}{\sin x \sin y}\\
\text{ctg } x - \text{ctg } y = \frac{\sin(y - x)}{\sin x \sin y}\\
\text{tg } x + \text{ctg } y = \frac{\cos(x - y)}{\cos x \sin y}\\
\text{ctg } x - \text{tg } y = \frac{\cos(x + y)}{\sin x \cos y}\\
2 \cos x \cos y = \cos(x + y) + \cos(x - y)\\
2 \sin x \sin y = \cos(x - y) - \cos(x + y)\\
2 \sin x \cos y = \sin(x + y) + \sin(x - y)\\
\end{align}
$$
# <mark class="hltr-coler">Формула вспомогательного угла</mark>
$$
\begin{align}
a^2+b^2 > 0\\
a\sin{x} + b\cos{x} = \sqrt{a^2 + b^2}
\left(
\frac{a}{ \underbrace{\sqrt{a^2 + b^2}}_{\overline{a}}}\sin{x}
+ \frac{b}{\underbrace{\sqrt{a^2 + b^2}}_{\overline{b}}}\cos{x}
\right) = \\
\overline{a}^2 + \overline{b}^2  = 1\Rightarrow
\exists \phi:(\overline{a},\overline{b}) = (\cos{\phi},\sin{\phi}) \Rightarrow\\
= \sqrt{a^2 + b^2}(\cos{\phi}\sin{x}) + \sin{\phi}\sin{x}) \\
= \sqrt{a^2 + b^2}(\sin{\phi + x})\\
\\
a\sin{x} + b\cos{x}= \sqrt{a^2 + b^2}(\sin{\phi + x})
\end{align}
$$
