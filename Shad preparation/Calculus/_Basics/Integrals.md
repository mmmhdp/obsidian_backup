# <mark class="hltr-coler">Таблица интегралов</mark>
$$
\begin{align}
\int 0  dx &= C \\
\int 1  dx &=x + C \\
\int x^n  dx &= \frac{x^{n+1}}{n+1} + C, x\neq -1, x>0\\
\int x^{-1}  dx &= \ln|x|+ C \\
\int a^x dx &= \frac{a^x}{\ln a} + C \\
\int e^x  dx &= e^x + C \\
\int \sin{x}  dx &= -\cos{x} + C \\
\int \cos{x}  dx &= \sin{x} + C \\
\int \frac{1}{\cos^2{x}}  dx &= \text{tg } x + C \\
\int \frac{1}{\sin^2{x}}  dx &= -\text{ctg } x + C \\
\int \frac{1}{\sqrt{a^2 - x^2}}  dx &= \text{arcsin }\frac{x}{a}+ C, |x|<|a| \\
\int \frac{1}{a^2+x^2}  dx &= \frac{1}{a}\text{arctg }\frac{x}{a} + C\\
\textit{"Высокий логарифм"}\\
\int \frac{1}{a^2 - x^2}  dx &= \frac{1}{2a}\ln|\frac{a+x}{a-x}| + C \\
\textit{"Длинный логарифм"}\\
\int \frac{1}{\sqrt{x^2\pm a^2}}  dx &= \ln \left|x+\sqrt{x^2\pm a^2}\right| + C \\
\end{align}
$$
